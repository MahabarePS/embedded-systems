#include<stdio.h>
#define ARRAY_SIZE 5
int main(void)
{
   int arr[ARRAY_SIZE]={0},i=0,element=0,place=0;

   printf("\nenter %d numbers  :  ",ARRAY_SIZE);
   for(i=0;i<ARRAY_SIZE;i++)
      scanf("%d",&arr[i]);



   for(i=1;i<ARRAY_SIZE;i++)
   {
       element = arr[i];
       for(place=i;place>0 ;place--)
       {
        if(element > arr[place-1])
           arr[place]=arr[place-1];
        else
           break;
       }

       arr[place]=element;

   }

   printf("\nSORTED ARRAY :  ");
   for(i=0;i<ARRAY_SIZE;i++)
      printf("%d\t",arr[i]);



   return 0;
}
