#include<stdio.h>
#define ARRAY_SIZE 5
int main(void)
{
   int arr[ARRAY_SIZE]={0},i=0,j=0,temp=0,flag=1;

   printf("\nenter %d numbers  :  ",ARRAY_SIZE);
   for(i=0;i<ARRAY_SIZE;i++)
      scanf("%d",&arr[i]);



   for(i=0;i<ARRAY_SIZE-1;i++)
   {
       for(flag=1,j=0;j<ARRAY_SIZE-1-i;j++)
       {
         if(arr[j]<arr[j+1])
         {
           temp = arr[j];
           arr[j]=arr[j+1];
           arr[j+1]=temp;
           flag=0;
         }
       }

       if(flag==1)
         break;
   }

   printf("\nSORTED ARRAY in %d passes :  ",i+1);
   for(i=0;i<ARRAY_SIZE;i++)
      printf("%d\t",arr[i]);



   return 0;
}
