#include<stdio.h>
#define ARRAY_SIZE 5
int main(void)
{
   int arr[ARRAY_SIZE]={0},i=0,j=0,temp=0,max=0;

   printf("\nenter %d numbers  :  ",ARRAY_SIZE);
   for(i=0;i<ARRAY_SIZE;i++)
      scanf("%d",&arr[i]);



   for(i=0;i<ARRAY_SIZE-1;i++)
   {
       max = i;
       for(j=i+1;j<ARRAY_SIZE;j++)
       {
         if(arr[max]<arr[j])
           max = j;
       }
       temp = arr[i];
       arr[i]=arr[max];
       arr[max]=temp;


   }

   printf("\nSORTED ARRAY :  ");
   for(i=0;i<ARRAY_SIZE;i++)
      printf("%d\t",arr[i]);



   return 0;
}
