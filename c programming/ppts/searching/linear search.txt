#include<stdio.h>
#define ARRAY_SIZE 5
int main(void)
{
   int arr[ARRAY_SIZE]={0},n=0,i=0;

   printf("\nenter %d numbers  :  ",ARRAY_SIZE);
   for(i=0;i<ARRAY_SIZE;i++)
      scanf("%d",&arr[i]);

   printf("\nenter number to be searched :  ");
   scanf("%d",&n);

   for(i=0;i<ARRAY_SIZE;i++)
      if(arr[i]==n)
          break;

   if(i==ARRAY_SIZE)
      printf("\n%d is not present in the array\n",n);
   else
      printf("\n%d found at index : %d \n",n,i);
   return 0;
}
