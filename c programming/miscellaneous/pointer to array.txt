#include <stdio.h>

int main(void) 
{
    int arr[5] = {10, 20, 30, 40, 50};


    int (*p)[5] = &arr;

    printf("arr is : %p\n", arr);
    printf("p is pointing to: %p\n", (*p));
    
    
    printf("First element: %d\n", (*p)[0]);
    printf("Third element: %d\n", (*p)[2]);
    printf("Last element: %d\n", (*p)[4]);
    printf("Last element: %d\n", *(*p+4));

    return 0;
}
