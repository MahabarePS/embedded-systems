#include "stm32f4xx_hal.h"
#include "string.h"
#include "stdio.h"

#define EEPROM_ADDR		0xA0
#define PAGE_SIZE		16

I2C_HandleTypeDef hi2c1;
UART_HandleTypeDef huart2;

static void I2C1_Init(void);
static void USART2_Init(void);


int main(void)
{
  char MSG1[] = "Writing Successful\n\r";
  char MSG2[] = "Reading Successful\n\r";
  char MSG3[] = "\n\r";
  char MSG4[] = "UnSuccessful Writing\n\r";
  char MSG5[] = "UnSuccessful Reading\n\r";
  uint8_t WriteData[] = "EEPROM Test";
  uint8_t ReadData[20];
  uint8_t Success;

  HAL_Init();
  I2C1_Init();
  USART2_Init();

  Success = HAL_I2C_Mem_Write (&hi2c1, EEPROM_ADDR, 0x10, 1, (uint8_t *) WriteData, strlen(WriteData), 100);
    HAL_Delay(10);
    if (Success == 0)
    {
  	  HAL_UART_Transmit(&huart2, (uint8_t *) MSG1, strlen(MSG1), 100);
    }
    else
    {
  	  HAL_UART_Transmit(&huart2, (uint8_t *) MSG4, strlen(MSG4), 100);
    }

    Success = HAL_I2C_Mem_Read (&hi2c1, EEPROM_ADDR, 0x10, 1, (uint8_t *) ReadData, 11, 100);
    if (Success == 0)
    {
  	  HAL_UART_Transmit(&huart2, (uint8_t *) MSG2, strlen (MSG2), 100);
  	  HAL_UART_Transmit(&huart2, (uint8_t *) ReadData, 11, 100);
  	  HAL_UART_Transmit(&huart2, (uint8_t *) MSG3, strlen (MSG3), 100);
    }
    else
      {
    	  HAL_UART_Transmit(&huart2, (uint8_t *) MSG5, strlen(MSG5), 100);
      }

  while(1)
  {

  }
}


static void I2C1_Init(void)
{
   GPIO_InitTypeDef GPIO_InitStruct = {0};


   __HAL_RCC_GPIOB_CLK_ENABLE();
   __HAL_RCC_I2C1_CLK_ENABLE();

	 GPIO_InitStruct.Pin = GPIO_PIN_8|GPIO_PIN_9;
	 GPIO_InitStruct.Mode = GPIO_MODE_AF_OD;
	 GPIO_InitStruct.Pull = GPIO_NOPULL;
	 GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_VERY_HIGH;
	 GPIO_InitStruct.Alternate = GPIO_AF4_I2C1;
	 HAL_GPIO_Init(GPIOB, &GPIO_InitStruct);

     hi2c1.Instance = I2C1;
     hi2c1.Init.ClockSpeed = 100000;
     hi2c1.Init.DutyCycle = I2C_DUTYCYCLE_2;
     hi2c1.Init.OwnAddress1 = 0;
     hi2c1.Init.AddressingMode = I2C_ADDRESSINGMODE_7BIT;
     hi2c1.Init.DualAddressMode = I2C_DUALADDRESS_DISABLE;
     hi2c1.Init.OwnAddress2 = 0;
     hi2c1.Init.GeneralCallMode = I2C_GENERALCALL_DISABLE;
     hi2c1.Init.NoStretchMode = I2C_NOSTRETCH_DISABLE;
     if (HAL_I2C_Init(&hi2c1) != HAL_OK)
     {
         // BUSINESS LOGIC
     }


}

static void USART2_Init(void)
{
   GPIO_InitTypeDef GPIO_InitStruct = {0};

   __HAL_RCC_USART2_CLK_ENABLE();
   __HAL_RCC_GPIOA_CLK_ENABLE();


	GPIO_InitStruct.Pin = GPIO_PIN_2|GPIO_PIN_3;
	GPIO_InitStruct.Mode = GPIO_MODE_AF_PP;
	GPIO_InitStruct.Pull = GPIO_NOPULL;
	GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_VERY_HIGH;
	GPIO_InitStruct.Alternate = GPIO_AF7_USART2;
	HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);

    huart2.Instance = USART2;
    huart2.Init.BaudRate = 9600;
    huart2.Init.WordLength = UART_WORDLENGTH_8B;
    huart2.Init.StopBits = UART_STOPBITS_1;
    huart2.Init.Parity = UART_PARITY_NONE;
    huart2.Init.Mode = UART_MODE_TX_RX;
    huart2.Init.HwFlowCtl = UART_HWCONTROL_NONE;
    huart2.Init.OverSampling = UART_OVERSAMPLING_16;
    if (HAL_UART_Init(&huart2) != HAL_OK)
    {
       // BUSINESS LOGIC
    }

}
