#include "stm32f4xx_hal.h"
#include "string.h"
#include "stdio.h"

#define EEPROM_WREN  0x06  /*!< Write Enable */
#define EEPROM_WRDI  0x04  /*!< Write Disable */
#define EEPROM_RDSR  0x05  /*!< Read Status Register */
#define EEPROM_WRSR  0x01  /*!< Write Status Register */
#define EEPROM_READ  0x03  /*!< Read from Memory Array */
#define EEPROM_WRITE 0x02  /*!< Write to Memory Array */

SPI_HandleTypeDef hspi1;

UART_HandleTypeDef huart2;

static void GPIO_Init(void);
static void SPI1_Init(void);
static void UART_Init(void);

static uint8_t RxBuffer[4];

static char MSG1[] = "SPI Writing Successful\n\r";
static char MSG2[] = "SPI Reading Successful\n\r";
static char MSG4[] = "UnSuccessful SPI Writing\n\r";
static char MSG5[] = "UnSuccessful SPI Reading\n\r";



void EEPROM_CS_LOW(void)
{
	HAL_GPIO_WritePin(GPIOB, GPIO_PIN_7, GPIO_PIN_RESET);
}

void EEPROM_CS_HIGH(void)
{
	 HAL_GPIO_WritePin(GPIOB, GPIO_PIN_7, GPIO_PIN_SET);
}

void SPI_EEPROM_WriteByte(unsigned int Addr, unsigned char Data)
{
	unsigned char Header[4],Success;

	EEPROM_CS_LOW();
	Header[0] = EEPROM_WREN;
	HAL_SPI_Transmit(&hspi1, (uint8_t*)Header, (uint16_t)1, 200);
	EEPROM_CS_HIGH();

	Header[0] = EEPROM_WRITE;
	Header[1] = Addr >> 8;
	Header[2] = Addr;
	Header[3] = Data;

	EEPROM_CS_LOW();
	Success = HAL_SPI_Transmit(&hspi1, (uint8_t*)Header, (uint16_t)4, 200);
	EEPROM_CS_HIGH();

    if (Success == 0)
    {
  	  HAL_UART_Transmit(&huart2, (uint8_t *) MSG1, strlen(MSG1), 100);
    }
    else
    {
  	  HAL_UART_Transmit(&huart2, (uint8_t *) MSG4, strlen(MSG4), 100);
    }

	HAL_Delay(20);
}

unsigned char SPI_EEPROM_ReadByte(unsigned int Addr)
{
	unsigned char Success,Header[4];
	unsigned char ReadData;

	Header[0] = EEPROM_READ;
	Header[1] = Addr>>8;
	Header[2] = Addr;


	EEPROM_CS_LOW();
    HAL_SPI_Transmit(&hspi1, (uint8_t*)Header, (uint16_t)3, 200);

    Success=HAL_SPI_Receive(&hspi1, (uint8_t*)&ReadData, (uint16_t)1, 200);
	EEPROM_CS_HIGH();

    if (Success == 0)
    {
  	  HAL_UART_Transmit(&huart2, (uint8_t *) MSG2, strlen(MSG1), 100);
    }
    else
    {
  	  HAL_UART_Transmit(&huart2, (uint8_t *) MSG5, strlen(MSG4), 100);
    }

	HAL_Delay(20);
	return ReadData;
}

int main(void)
{

  HAL_Init();
  GPIO_Init();
  SPI1_Init();
  UART_Init();


  SPI_EEPROM_WriteByte(0x10, 'A');
  HAL_Delay(10);
  RxBuffer [0] = SPI_EEPROM_ReadByte(0x10);
  HAL_Delay(10);
  SPI_EEPROM_WriteByte(0x10, 'B');
  HAL_Delay(10);
  RxBuffer [1] = SPI_EEPROM_ReadByte(0x10);

  while (1)
  {

  }

}


static void SPI1_Init(void)
{
   GPIO_InitTypeDef GPIO_InitStruct = {0};

   __HAL_RCC_SPI1_CLK_ENABLE();

   __HAL_RCC_GPIOA_CLK_ENABLE();
    /**SPI1 GPIO Configuration
    PA5     ------> SPI1_SCK
    PA6     ------> SPI1_MISO
    PA7     ------> SPI1_MOSI
    */
    GPIO_InitStruct.Pin = GPIO_PIN_5|GPIO_PIN_6|GPIO_PIN_7;
    GPIO_InitStruct.Mode = GPIO_MODE_AF_PP;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_VERY_HIGH;
    GPIO_InitStruct.Alternate = GPIO_AF5_SPI1;
    HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);

  hspi1.Instance = SPI1;
  hspi1.Init.Mode = SPI_MODE_MASTER;
  hspi1.Init.Direction = SPI_DIRECTION_2LINES;
  hspi1.Init.DataSize = SPI_DATASIZE_8BIT;
  hspi1.Init.CLKPolarity = SPI_POLARITY_LOW;
  hspi1.Init.CLKPhase = SPI_PHASE_1EDGE;
  hspi1.Init.NSS = SPI_NSS_SOFT;
  hspi1.Init.BaudRatePrescaler = SPI_BAUDRATEPRESCALER_32;
  hspi1.Init.FirstBit = SPI_FIRSTBIT_MSB;


  HAL_SPI_Init(&hspi1);

}

static void UART_Init(void)
{
  GPIO_InitTypeDef GPIO_InitStruct = {0};

  __HAL_RCC_USART2_CLK_ENABLE();

  __HAL_RCC_GPIOA_CLK_ENABLE();
    /**USART2 GPIO Configuration
    PA2     ------> USART2_TX
    PA3     ------> USART2_RX
    */
   GPIO_InitStruct.Pin = GPIO_PIN_2|GPIO_PIN_3;
   GPIO_InitStruct.Mode = GPIO_MODE_AF_PP;
   GPIO_InitStruct.Pull = GPIO_NOPULL;
   GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_VERY_HIGH;
   GPIO_InitStruct.Alternate = GPIO_AF7_USART2;
   HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);



  huart2.Instance = USART2;
  huart2.Init.BaudRate = 9600;
  huart2.Init.WordLength = UART_WORDLENGTH_8B;
  huart2.Init.StopBits = UART_STOPBITS_1;
  huart2.Init.Parity = UART_PARITY_NONE;
  huart2.Init.Mode = UART_MODE_TX_RX;
  huart2.Init.HwFlowCtl = UART_HWCONTROL_NONE;
  huart2.Init.OverSampling = UART_OVERSAMPLING_16;

  HAL_UART_Init(&huart2);

}

static void GPIO_Init(void)
{
  GPIO_InitTypeDef GPIO_InitStruct = {0};

  __HAL_RCC_GPIOA_CLK_ENABLE();
  __HAL_RCC_GPIOB_CLK_ENABLE();


  HAL_GPIO_WritePin(GPIOB, GPIO_PIN_7, GPIO_PIN_RESET);

  /*Configure GPIO pin : PB7 */
  GPIO_InitStruct.Pin = GPIO_PIN_7;
  GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
  GPIO_InitStruct.Pull = GPIO_NOPULL;
  GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_LOW;
  HAL_GPIO_Init(GPIOB, &GPIO_InitStruct);

}
